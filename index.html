  <script src="https://aframe.io/releases/1.2.0/aframe.min.js"></script>

<body style="margin: 0; overflow: hidden;">


  <!-- ARの撮影画面を作成 -->

  <a-scene embedded arjs="sourceType: webcam; debugUIEnabled: false;">

    

    <!-- Hiroマーカー（カメラにかざすマーク）を認識した時の処理 -->

    <a-marker preset="hiro">

      <!-- ここにスプレッドシートの文字を表示する（初期値は「読み込み中...」） -->

      <a-text id="ar-text" value="Loading..." position="0 0.5 0" rotation="-90 0 0" color="red" align="center" scale="2 2 2"></a-text>

    </a-marker>


    <!-- カメラの配置 -->

    <a-entity camera></a-entity>

  </a-scene>


  <script>

    // ★ここに手順1でコピーしたGASのURLを貼り付けます★

    const GAS_URL = "https://script.google.com/a/macros/batonz.co.jp/s/AKfycbyqanQoAAmCqdtUDyYH3fmLrZmAMCOAWkZeuBKcQW664QoyPQhvDb3xdEI3pLwGnW-hcQ/exec";


    // GASからデータを取ってくる処理（fetch）

    fetch(GAS_URL)

      .then(response => response.json())

      .then(data => {

        // 例: スプレッドシートのA1セル・B1セルの文字を取得して繋げる

        // data[0][0] がA1セル、data[0][1] がB1セルに入っている文字です

        const textToShow = data[0][0] + " : " + data[0][1];

        

        // 画面上のARテキストを書き換える

        document.getElementById("ar-text").setAttribute("value", textToShow);

      })

      .catch(error => {

        console.error("エラー:", error);

        document.getElementById("ar-text").setAttribute("value", "Error!");

      });

  </script>


</body>
